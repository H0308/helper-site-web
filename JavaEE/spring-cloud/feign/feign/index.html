// 在注册中心中找value对应的服务，path表示请求的统一前缀
@FeignClient(value = "product-service", path = "/product")
public interface ProductApi {
    // 通过id查询
    @RequestMapping("/{id}")
    Product getProductById(@PathVariable Integer id);

    // 接收一个参数，不能省略@RequestParam
    @RequestMapping("/p1")
    String p1(@RequestParam("id") Integer id);

    // 接收多个参数
    @RequestMapping("/p2")
    String p2(@RequestParam("id") Integer id, @RequestParam("name") String name);

    // 接收一个对象
    // OpenFeign客户端接收一个对象需要使用@SpringQueryMap
    @RequestMapping("/p3")
    String p3(@SpringQueryMap Product product);

    // 接收一个JSON
    @RequestMapping("/p4")
    String p4(@RequestBody Product product);
}

@Service
public class OrderService {
    @Autowired
    private OrderMapper orderMapper;

    @Autowired
    private RestTemplate restTemplate;
    @Autowired
    private ProductApi productApi;

    public Order getOrderById(Integer id) {
        Order order = orderMapper.selectOrderById(id);
        Product product = productApi.getProductById(order.getProductId());
        order.setProduct(product);
        return order;
    }

    // 接收一个参数
    public String p1(Integer id) {
        return productApi.p1(id);
    }

    // 接收多个参数
    public String p2(Integer id, String name) {
        return productApi.p2(id, name);
    }

    // 接收一个对象
    public String p3(Product product) {
        return productApi.p3(product);
    }

    // 接收一个JSON
    public String p4(Product product) {
        return productApi.p4(product);
    }
}

OpenFeign最佳实践¶

下面的两种方式，都会创建一个单独的API模块，其他模块要使用API模块，就需要先将API模块使用Maven的install先把API模块打包安装到本地仓库

如果是服务器部署并且不使用Maven中央仓库或者私服的情况下，则可以考虑本地读取：

  1. 在开发环境下先使用Maveninstall把API模块打包安装到本地仓库

  2. 修改引入了API模块的其他模块的pom文件，在依赖配置中添加如下的代码，然后再打包该模块：

    仓库查找依赖 注意使用斜杠而不是反斜杠-->
    <scope>system</scope>

<build>

        <plugin>
            <groupId>org.springframework.boot</groupId>
            <artifactId>spring-boot-maven-plugin</artifactId>
            <!-- 打包时就会把上面那个本地的 product-api.jar 一并塞进最终的可执行 jar 里 -->
            <configuration>
                <includeSystemScope>true</includeSystemScope>
            </configuration>
        </plugin>
    </plugins>
</build>

  3. 将可执行的jar上传到服务器即可，此时不需要再考虑API模块，因为其打包结果已经塞进可执行的jar中了

public class ProductController implements ProductInterface {

    public Product getProduct(@PathVariable Integer id) {
